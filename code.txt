<button onclick="prevSlide()" style="position:absolute; left:10px; top:50%; z-index:10;">❮</button>
<button onclick="nextSlide()" style="position:absolute; right:10px; top:50%; z-index:10;">❯</button>

<!-- JavaScript executed through onload to bypass Shopify blocking -->
<div style="display:none"
     onload="
     window.myslideIndex = 0;
     window.myslides = document.querySelectorAll('#myslideshow .myslide');
     window.showSlide = function(i){
       myslides[myslideIndex].style.display='none';
       myslideIndex = (i + myslides.length) % myslides.length;
       myslides[myslideIndex].style.display='block';
     };
     window.nextSlide = ()=>showSlide(myslideIndex+1);
     window.prevSlide = ()=>showSlide(myslideIndex-1);
     setInterval(()=>nextSlide(), 4000);
     ">
</div>
